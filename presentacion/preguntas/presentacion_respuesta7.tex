\begin{frame}{El proceso en cuatro etapas}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{1. Acondicionamiento}
                \begin{itemize}
                    \item Residuos org\'{a}nicos a 60--70\% humedad.
                    \item Relaci\'{o}n C:N optimizada.
                    \item 700--800 huevos por reactor (prototipo).
                \end{itemize}
            \end{block}
            \begin{block}{2. Crecimiento larvario}
                \begin{itemize}
                    \item Proteasas, lipasas, celulasas.
                    \item Microbiota: \textit{Enterococcus}, \textit{Bacillus}.
                    \item Biomasa: 180--250 mg/individuo.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{3. Separaci\'{o}n}
                \begin{itemize}
                    \item Prepupas migran por rampas.
                    \item Auto-recolecci\'{o}n sin intervenci\'{o}n invasiva.
                    \item Frass como fertilizante estabilizado.
                \end{itemize}
            \end{block}
            \begin{block}{4. Cierre del ciclo}
                \begin{itemize}
                    \item Pupaci\'{o}n: 10--14 d\'{i}as.
                    \item Adultos: iluminaci\'{o}n UV-A, 5--8 d\'{i}as.
                    \item 200--900 huevos/hembra para reinicio.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
