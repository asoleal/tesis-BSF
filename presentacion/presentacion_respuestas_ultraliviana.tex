\begin{frame}{Pregunta 1: El sistema como metabolismo acoplado}
\begin{block}{Condiciones óptimas}
\footnotesize
27--30\,°C, HR 60--70\%, aireación suficiente $\rightarrow$ proceso predominantemente aerobio.
\end{block}
\begin{exampleblock}{CO$_2$ como señal fisiológica}
\footnotesize
No es solo emisión: refleja costos de mantenimiento, síntesis de biomasa y acumulación de lípidos.
\end{exampleblock}
\end{frame}

\begin{frame}{Pregunta 1: CH$_4$ --- ¿De dónde sale?}
\begin{alertblock}{No es metabolismo larval}
\footnotesize
Es producto de microambientes anóxicos en el sustrato, generados por arqueas metanogénicas estrictas.
\end{alertblock}
\begin{block}{Causas de anoxia}
\footnotesize
\begin{itemize}
    \item Humedad $>80$\% (saturación de poros)
    \item Compactación por alta densidad larval
    \item Sustrato fino que bloquea difusión de O$_2$
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Pregunta 1: CH$_4$ --- Dos rutas bioquímicas}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Ruta acetoclástica}
\centering
\footnotesize
acetato $\rightarrow$ CH$_4$ + CO$_2$
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Ruta hidrogenotrófica}
\centering
\footnotesize
CO$_2$ = aceptor de electrones
\end{block}
\end{column}
\end{columns}
\vspace{0.3cm}
\centering
\footnotesize Ambas operan en microambientes donde el O$_2$ no llega.
\end{frame}

\begin{frame}{Pregunta 1: N$_2$O --- Nitrificación}
\begin{block}{Primer paso: oxidación aerobia}
\footnotesize
Bacterias \textit{Nitrosomonas} spp. transforman NH$_4^+$ en NO$_2^-$ y NO$_3^-$.
\end{block}
\begin{exampleblock}{Requisito}
\footnotesize
Oxígeno disponible en el sustrato.
\end{exampleblock}
\end{frame}

\begin{frame}{Pregunta 1: N$_2$O --- Denitrificación y escape}
\begin{alertblock}{Segundo paso: reducción anaerobia}
\footnotesize
Bacterias facultativas reducen NO$_3^-$ pasando por NO $\rightarrow$ \textbf{N$_2$O} $\rightarrow$ N$_2$.
\end{alertblock}
\begin{alertblock}{¿Por qué se escapa N$_2$O?}
\footnotesize
Denitrificación incompleta por:
\begin{itemize}
    \item Limitación de sustrato carbonado
    \item Variaciones bruscas de oxigenación
    \item Exceso de nitrógeno mineralizable
\end{itemize}
\end{alertblock}
\end{frame}

\begin{frame}{Pregunta 1: Papel físico de las larvas}
\begin{block}{Bioturbación positiva}
\footnotesize
Fragmentan residuo, modifican estructura porosa, reducen tiempo en zonas anóxicas. Incorporan N a su biomasa, disminuyendo N libre para N$_2$O.
\end{block}
\begin{alertblock}{Límite de la bioturbación}
\footnotesize
Densidad excesiva $\rightarrow$ compactación $\rightarrow$ microambientes anóxicos $\rightarrow$ CH$_4$ y N$_2$O.
\end{alertblock}
\end{frame}

\begin{frame}{Pregunta 1: BSF vs. Compostaje Tradicional}
\begin{exampleblock}{Ventaja potencial}
\footnotesize
BSF bien manejada: \textbf{60--80\% menos CH$_4$} que compostaje mal aireado.
\end{exampleblock}
\begin{alertblock}{Riesgo de greenwashing involuntario}
\footnotesize
Bajo exceso de humedad, alta densidad o sustrato lignocelulósico de baja degradabilidad, BSF puede igualar o superar emisiones del compostaje.
\end{alertblock}
\end{frame}

\begin{frame}{Pregunta 1: Variable crítica --- Humedad}
\centering
\begin{tikzpicture}

\normalsize

};
\end{tikzpicture}
\end{frame}

\begin{frame}{Pregunta 2: El problema de los factores estáticos}
\begin{alertblock}{Supuesto falso}
\footnotesize
Los factores estáticos asumen que el proceso siempre se comporta igual.
\end{alertblock}
\begin{block}{Realidad de \BSF}
\footnotesize
Emisiones cambian cada hora según humedad, densidad larval, temperatura y oxígeno.
\end{block}
\begin{exampleblock}{Consecuencia}
\footnotesize
Promedios fijos del IPCC o compostaje generan errores grandes, subestiman picos de metano y distorsionan balances de carbono.
\end{exampleblock}
\end{frame}

\end{exampleblock}

\begin{frame}{Pregunta 3: Límite Estructural de Eriksen}
\begin{alertblock}{Supuesto de homogeneidad}
\footnotesize
Modelo asume larvas uniformes y condiciones macroscópicas estables.
\end{alertblock}
\begin{block}{Realidad}
\footnotesize
Agregaciones dinámicas $\rightarrow$ fluidos activos $\rightarrow$ gradientes 3D de T y O$_2$.
\end{block}
\end{frame}

\begin{frame}{Pregunta 3: Desviaciones Sistemáticas}
\begin{block}{Atribución errónea}
\footnotesize
El modelo deterministico atribuye desviaciones a ``error paramétrico''.
\end{block}
\begin{alertblock}{Causa real}
\footnotesize
Son \textbf{limitación arquitectónica estructural}: parámetros concentrados no capturan heterogeneidad espacial.
\end{alertblock}
\end{frame}

\begin{frame}{Pregunta 3: Corrección --- Sensor Virtual}
\begin{exampleblock}{Red feedforward}
\footnotesize
Estima $B(t)$ desde variables observables:
\begin{itemize}
    \item Densidad inicial
    \item Temperatura
    \item Humedad
\end{itemize}
\end{exampleblock}
\begin{block}{Ventaja}
\footnotesize
Resuelve problema inverso sin muestreos destructivos.
\end{block}
\end{frame}

\begin{frame}{Pregunta 3: Corrección --- PINN}
\begin{block}{Physics-Informed Neural Network}
\footnotesize
Resuelve EDO de Eriksen con parámetros de red como incógnitas.
\end{block}
\begin{exampleblock}{Restricción dura}
\footnotesize
Pérdida = error de ajuste a sensores + residual de la EDO. Garantiza conservación de masa.
\end{exampleblock}
\end{frame}

\begin{frame}{Pregunta 3: ¿Por qué CO$_2$?}
\begin{block}{Gas primario}
\footnotesize
Respiración aerobia directamente vinculada a dinámica de $B(t)$.
\end{block}
\begin{exampleblock}{Condición necesaria}
\footnotesize
Medición continua para validar estado metabólico y calibrar modelo híbrido.
\end{exampleblock}
\end{frame}

\begin{frame}{Pregunta 3: ¿Por qué CH$_4$?}
\begin{alertblock}{Señal de fallo operativo}
\footnotesize
No es metabolismo larval. Indica activación de arqueas en microambientes anóxicos.
\end{alertblock}
\begin{block}{Relevancia climática}
\footnotesize
GWP 27,2$\times$ CO$_2$ (AR6) = mayor impacto en balances de corto plazo.
\end{block}
\end{frame}

\begin{frame}{Pregunta 3: ¿Por qué no N$_2$O?}
\begin{block}{Decisión metodológica explícita}
\footnotesize
No es omisión. Requiere:
\begin{itemize}
    \item Sensores electroquímicos específicos
    \item Diseños experimentales diferenciados
    \item Cadena de calibración independiente
\end{itemize}
\end{block}
\begin{alertblock}{Fuera del alcance}
\footnotesize
Esta fase no valida el ciclo del nitrógeno.
\end{alertblock}
\end{frame}

\begin{alertblock}{Deriva en el tiempo}
\footnotesize
\begin{itemize}
    \item Envejecimiento de componentes ópticos
    \item Fluctuaciones térmicas
    \item Acumulación de polvo orgánico
\end{itemize}
\end{alertblock}

\footnotesize
Error sistemático que se acumula en mediciones prolongadas.

\begin{frame}{Pregunta 4: Fuente de Error 2 --- Interferencias}
\begin{alertblock}{Señales falsas}

    \item Humedad relativa elevada
    \item NH$_3$ del sustrato
    \item Compuestos orgánicos volátiles
\end{itemize}
\end{alertblock}
\begin{block}{Efecto}
\footnotesize
Absorbancia alterada $\rightarrow$ lecturas falsamente altas o bajas.

\begin{block}{Fuera del rango nominal}
\footnotesize
Respuesta no lineal en concentraciones extremas.
\end{block}
\begin{alertblock}{Limitación temporal}
\footnotesize
Capacidad insuficiente para seguir picos rápidos de CO$_2$ o CH$_4$.

\begin{frame}{Pregunta 5: Desafío 1 --- Parametrización}
\begin{alertblock}{Especificidad fisiológica}
\footnotesize
Eficiencia de asimilación, costo de mantenimiento y coeficiente respiratorio dependen de genética, edad e historial térmico.
\end{alertblock}
\begin{block}{Variabilidad del sustrato}
\footnotesize
Residuos agroindustriales heterogéneos alteran cinética de ingestión y digestión.
\end{block}
\end{frame}

\begin{frame}{Pregunta 5: Desafío 2 --- Microbiota Oculta}
\begin{alertblock}{Compartimento desconocido}
\footnotesize
Fracción indeterminada del CO$_2$ emitido proviene de la respiración de la microbiota del sustrato, no del insecto.
\end{alertblock}
\begin{block}{Implicación}
\footnotesize
Discriminar fuentes requiere diseños experimentales con tratamientos específicos.
\end{block}
\end{frame}

\begin{frame}{Pregunta 5: Desafío 3 --- Validación}
\begin{alertblock}{Muestreo destructivo}
\footnotesize
Interrumpe el reactor y altera la dinámica de competencia y aireación.
\end{alertblock}
\begin{block}{Conversión concentración $\rightarrow$ tasa}
\footnotesize
Requiere flujo de arrastre, presión y temperatura del gas con precisión.
\end{block}
\end{frame}

\begin{frame}{Pregunta 5: Desafío 4 --- Heterogeneidad Espacial}
\begin{alertblock}{Agregaciones larvales}
\footnotesize
Se comportan como fluidos activos, generando gradientes tridimensionales de T y O$_2$.
\end{alertblock}
\begin{block}{Limitación del modelo}
\footnotesize
EDO de orden cero no captura heterogeneidad espacial. El residual sistemático es arquitectónico, no paramétrico.
\end{block}
\end{frame}

\begin{frame}{Pregunta 8: Naturaleza del Mecanicista}
\begin{block}{Base}
\footnotesize
Leyes de conservación de masa y energía.
\end{block}
\begin{exampleblock}{Formulación}
\footnotesize
EDO para $B(t)$: asimilación neta menos costos respiratorios.
\end{exampleblock}

\begin{frame}{Pregunta 8: Ecuación del Mecanicista}
\centering
\Large
$R_{\text{CO}_2} = Y_{\text{CO}_2} \cdot \frac{dB}{dt} + m \cdot B(t)$
\vspace{0.3cm}

\footnotesize
Crecimiento + Mantenimiento
\end{frame}

\begin{frame}{Pregunta 8: Alcances del Mecanicista}
\begin{exampleblock}{Fortalezas}
\footnotesize
\begin{itemize}
    \item Interpretabilidad biológica directa
    \item Pocos datos de entrada una vez calibrado
    \item Línea base físicamente consistente
    \item Comunicación clara con auditores
\end{itemize}
\end{exampleblock}
\end{frame}

\begin{frame}{Pregunta 8: Limitación 1 --- Homogeneidad}
\begin{alertblock}{Supuesto}
\footnotesize
Medio perfectamente mezclado, parámetros uniformes.
\end{alertblock}
\begin{block}{Realidad}
\footnotesize
Agregaciones larvales generan gradientes locales de T, HR y O$_2$.
\end{block}
\end{frame}

\begin{frame}{Pregunta 8: Limitación 2 --- Aerobiosis Exclusiva}
\begin{alertblock}{Ceguera estructural}
\footnotesize
No representa rutas fermentativas paralelas (metanogénesis).
\end{alertblock}
\begin{block}{Consecuencia}
\footnotesize
Ante microambientes anóxicos, sigue prediciendo solo CO$_2$ de respiración aerobia.
\end{block}
\end{frame}

\begin{frame}{Pregunta 8: Limitación 3 --- Rigidez}
\begin{block}{Parámetros no universales}
\footnotesize
Calibrados en dieta estándar de laboratorio.
\end{block}
\begin{alertblock}{Transferibilidad}
\footnotesize
Cambio a residuo agroindustrial heterogéneo requiere recalibración completa.
\end{alertblock}
\end{frame}

\begin{frame}{Pregunta 8: Híbrido --- Sensor Virtual}
\begin{exampleblock}{Función}
\footnotesize
Red neuronal densa estima $B(t)$ desde observables (densidad inicial, T, HR).
\end{exampleblock}
\begin{block}{Beneficio}
\footnotesize
Sin muestreos destructivos que interrumpan el reactor.
\end{block}
\end{frame}

\begin{frame}{Pregunta 8: Híbrido --- PINN}
\begin{block}{Mecanismo}
\footnotesize
Resuelve EDO con pesos de red como incógnitas.
\end{block}
\begin{exampleblock}{Restricción}
\footnotesize
Espacio de soluciones acotado a trayectorias físicamente admisibles.
\end{exampleblock}
\end{frame}

\begin{frame}{Pregunta 8: Híbrido --- Ventajas}
\begin{exampleblock}{Precisión}
\footnotesize
Mayor ajuste en condiciones reales de campo.
\end{exampleblock}
\begin{exampleblock}{Adaptabilidad}
\footnotesize
No requiere recalibración completa ante cambios de sustrato.
\end{exampleblock}
\begin{exampleblock}{Detección}
\footnotesize
Alerta de anaerobiosis por residual CH$_4$ medido vs. predicho.
\end{exampleblock}
\end{frame}

\begin{frame}{Pregunta 8: Híbrido --- Limitaciones}
\begin{alertblock}{Dependencia de datos}
\footnotesize
Errores instrumentales o deriva no corregida se propagan a predicciones.
\end{alertblock}
\begin{block}{Costo computacional}
\footnotesize
Mayor que la solución numérica de una EDO simple.
\end{block}
\begin{alertblock}{Interpretabilidad}
\footnotesize
Pesos de la red no tienen traducción biológica inmediata.
\end{alertblock}
\end{frame}

\begin{frame}{Pregunta 9: Ruralidad}

\begin{frame}{Pregunta 9: Dependencia del Sistema}

Sin soporte técnico local, el operador depende de asistencia urbana demorada.
\end{alertblock}
\begin{exampleblock}{Alternativa}
\footnotesize
Redes LoRa/LoRaWAN o SIM M2M para mantener canal de comunicación.
\end{exampleblock}
\end{frame}

\begin{frame}{Pregunta 9: Electricidad}
\begin{alertblock}{Cortes frecuentes}
\footnotesize
Horas o días sin suministro en zonas rurales colombianas.
\end{alertblock}
\begin{block}{Consecuencia}
\footnotesize
Sensores dejan de medir, ventiladores se apagan.
\end{block}
\begin{alertblock}{Riesgo biológico}
\footnotesize
Pocas horas sin ventilación $\rightarrow$ anaerobiosis $\rightarrow$ mortalidad masiva.
\end{alertblock}
\end{frame}

\begin{frame}{Pregunta 9: Mitigación Energética}
\begin{exampleblock}{Solución}
\footnotesize
Paneles solares fotovoltaicos + baterías de respaldo.
\end{exampleblock}
\begin{block}{Resultado}
\footnotesize
Continuidad de monitoreo y ventilación durante cortes prolongados.
\end{block}
\end{frame}

\begin{frame}{Pregunta 9: Ciberseguridad}
\begin{alertblock}{Amenaza}
\footnotesize
IoT con IP expuesta es vulnerable a intrusión.
\end{alertblock}
\begin{block}{Medidas}
\footnotesize
\begin{itemize}
    \item Cifrado TLS 1.2+
    \item Autenticación X.509
    \item Segmentación VLAN / firewall
    \item Firmware firmado criptográficamente
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Pregunta 9: Integridad MRV}
\begin{alertblock}{Cadena de custodia}
\footnotesize
Condición necesaria para validez de créditos de carbono (VCS, Gold Standard).
\end{alertblock}
\begin{block}{Consecuencia de manipulación}
\footnotesize
Violación de certificación con responsabilidades legales directas.
\end{block}
\end{frame}

\begin{frame}{Pregunta 9: Calidad de Datos}
\begin{alertblock}{Deriva y suciedad}
\footnotesize
Polvo orgánico y humedad saturan sensores de bajo costo.
\end{alertblock}
\begin{block}{Efecto en IA}
\footnotesize
Red neuronal entrenada con datos falsos aprende patrones falsos.
\end{block}
\begin{alertblock}{Riesgo operativo}
\footnotesize
Operador confía en predicciones erradas y deja de inspeccionar visualmente.
\end{alertblock}
\end{frame}

\begin{frame}{Pregunta 9: Diseño Robusto}
\begin{exampleblock}{Estrategia propuesta}
\footnotesize
\begin{itemize}
    \item ESP32 de bajo costo
    \item Interfaces I2C/UART estándar
    \item Software libre (Python, TensorFlow)
    \item Almacenamiento local ante fallos
    \item Calibración periódica vs. referencia
\end{itemize}
\end{exampleblock}
\end{frame}
